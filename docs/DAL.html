 35class DAL_Procedure_Engine:
 36    def __init__(self, connection, cursor):
 37        self.connection = connection
 38        self.cursor = cursor
 39    
 40    def call_procedure(self, procedure_name, params = None):
 41        if self.cursor is None:
 42            return None, None
 43        
 44        parameters = params or []
 45        
 46        try:
 47            self.cursor.callproc(procedure_name, parameters)
 48            for result in self.cursor.stored_results():
 49                headers = [column[0] for column in result.description]
 50                records = result.fetchall()
 51                return headers, records
 52            logger.warning(f"Procedure '{procedure_name}' returned no result sets")
 53            return None, None
 54        except mysql.connector.Error as e:
 55            logger.error(f"Failed to call procedure '{procedure_name}': {e}")
 56            return None, None
 57    
 58    def call_procedure_no_results(self, procedure_name, params = None):
 59        if self.cursor is None or self.connection is None:
 60            return False
 61        
 62        parameters = params or []
 63        try:
 64            self.cursor.callproc(procedure_name, parameters)
 65            self.connection.commit()
 66            return True
 67        except mysql.connector.Error as e:
 68            logger.error(f"Failed to call procedure '{procedure_name}': {e}")
 69            self.connection.rollback()
 70            return False
